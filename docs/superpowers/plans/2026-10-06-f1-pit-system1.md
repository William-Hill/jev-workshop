# F1 Pit Decision: System 1 → System 2 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking. Agents without these skills (for example Devin): ignore this line and follow the tasks in order.

**Goal:** Measure Jev on a real labeled decision, "should this driver pit at the end of this lap?", on historical F1 races from FastF1, against a hand-written rule and a trained logistic regression, with an LLM as System 2 for uncertain laps.

**Architecture:** A Python package `f1/f1pit/`. `data.py` turns a FastF1 race into one row for each driver and lap; `facts.py` turns a row into the facts that code calculates; `baselines.py`, `jev.py`, and `system2.py` give a pit probability; `metrics.py` scores them against the teams' real stops; `run.py` is the command line. Train and tune on 2023, report on 2024.

**Tech Stack:** Python 3.11+, `fastf1`, `pandas`, `scikit-learn`, `typesafe-sdk` (`TypeSafeClient`, `AsyncTypeSafeClient`, `Noul`, `NoulCriteria`), `pytest`; System 2 through `anthropic` or headless Claude Code (`claude -p`).

- A lap with no car behind (the last car). Expected: `gap_behind_s` is `None`, and `stop_now_loses_position` is `false`, not an error. Test in Task 2.
- A wet race (intermediate or wet tires). Expected: `stop_still_required` is `false`, because the two-compound rule does not apply. Test in Task 3.
- A lap with a red flag. Expected: the lap is removed from the table. Test in Task 2.

- The LLM answers with text that is not "yes" or "no". Expected: System 2 gives `None`, and the Jev probability stays. Test in Task 6.

```python
"""Turn one FastF1 race into one row for each driver and lap.

The label is the team's real decision: pitted = the car entered the pit lane on this lap.
"""
import os
import pandas as pd
from .config import CACHE_DIR, DATA_DIR

DEFAULT_PIT_LOSS_S = 22.0  # Used only when a race has no clean stop to measure.


def build_lap_table(session) -> pd.DataFrame:
    laps = session.laps
    df = pd.DataFrame({
        "driver": laps["Driver"].astype(str),
        "lap": laps["LapNumber"].astype(int),
        "end_s": laps["Time"].dt.total_seconds(),
        "lap_time_s": laps["LapTime"].dt.total_seconds(),
        "stint": laps["Stint"],
        "compound": laps["Compound"].fillna("UNKNOWN").astype(str),
        "tyre_life": laps["TyreLife"],
        "position": laps["Position"],
        "track_status": laps["TrackStatus"].fillna("").astype(str),
        "pitted": laps["PitInTime"].notna(),
        "pit_out": laps["PitOutTime"].notna(),
    })
    total = int(df["lap"].max())
    df["laps_remaining"] = total - df["lap"]
    df["safety_car"] = df["track_status"].str.contains("4")
    df["vsc"] = df["track_status"].str.contains("6")
    df["red"] = df["track_status"].str.contains("5")

    # The gap to the car behind at the end of the lap.
    df = df.sort_values(["lap", "end_s"])
    df["gap_behind_s"] = df.groupby("lap")["end_s"].shift(-1) - df["end_s"]

    # Stops done before this lap.
    df = df.sort_values(["driver", "lap"])
    df["stops_done"] = df.groupby("driver")["pitted"].cumsum() - df["pitted"].astype(int)

    # Rain in the 2 minutes before the end of the lap.
    w = session.weather_data
    rain_s = w.loc[w["Rainfall"].astype(bool), "Time"].dt.total_seconds() if len(w) else pd.Series(dtype=float)
    df["rain"] = df["end_s"].apply(lambda t: bool(((rain_s > t - 120) & (rain_s <= t)).any()))

    # Race-control messages of this lap (text).
    m = session.race_control_messages
    msgs = m.dropna(subset=["Lap"]).groupby(m["Lap"].dropna().astype(int))["Message"].apply(list).to_dict() if len(m) and "Lap" in m else {}
    df["messages"] = df["lap"].map(lambda lap: msgs.get(lap, []))

    df = df[(df["lap"] > 1) & (df["laps_remaining"] > 0) & (~df["red"])]
    return df.reset_index(drop=True)


def pit_loss_s(table: pd.DataFrame) -> float:
    """Median time lost by a stop: in-lap + out-lap minus two typical green laps of that driver."""
    green = table[~table["safety_car"] & ~table["vsc"] & ~table["pitted"] & ~table["pit_out"]].groupby("driver")["lap_time_s"].median()
    losses = []
    for drv, g in table.groupby("driver"):
        g = g.set_index("lap")
        for lap in g.index[g["pitted"]]:
            if lap + 1 in g.index and bool(g.loc[lap + 1, "pit_out"]) and not bool(g.loc[lap, "safety_car"]) and drv in green:
                losses.append(g.loc[lap, "lap_time_s"] + g.loc[lap + 1, "lap_time_s"] - 2 * green[drv])
    s = pd.Series(losses, dtype=float).dropna()
    return float(s.median()) if len(s) else DEFAULT_PIT_LOSS_S


def typical_stints(tables) -> dict:
    """Median length of completed stints (stints that end with a stop), for each compound."""
    lengths = []
    for t in tables:
        done = t[t["pitted"]]
        for _, r in done.iterrows():
            lengths.append({"compound": r["compound"], "length": int(r["tyre_life"])})
    df = pd.DataFrame(lengths)
    return {} if df.empty else {k: int(v) for k, v in df.groupby("compound")["length"].median().items()}


def load_race(year: int, gp: str) -> pd.DataFrame:
    """Load a race from FastF1 (cached), build the table, and keep a copy in f1/data/."""
    import fastf1
    os.makedirs(CACHE_DIR, exist_ok=True)
    os.makedirs(DATA_DIR, exist_ok=True)
    # Pickle is safe here: this code writes these files itself in f1/data/ (not committed). Never load a pickle from elsewhere.
    path = os.path.join(DATA_DIR, f"{year}-{gp.replace(' ', '_')}.pkl")
    if os.path.exists(path):
        return pd.read_pickle(path)
    fastf1.Cache.enable_cache(CACHE_DIR)
    s = fastf1.get_session(year, gp, "R")
    s.load(laps=True, weather=True, messages=True, telemetry=False)
    t = build_lap_table(s)
    t["race"] = f"{year} {gp}"
    t.to_pickle(path)
    return t
```

```python
"""The facts that code calculates for one driver and lap. Jev gets these, never raw timing."""
import math

WET = ("INTERMEDIATE", "WET")


def _num(x):
    return None if x is None or (isinstance(x, float) and math.isnan(x)) else x


def lap_facts(r, typical: dict, pit_loss: float) -> dict:
    t = typical.get(r["compound"])
    age = _num(r["tyre_life"])
    if t is None or age is None:
        age_vs = "unknown"
    elif age > t:
        age_vs = "past"
    elif age >= t - 3:
        age_vs = "near"
    else:
        age_vs = "well before"
    gap = _num(r["gap_behind_s"])
    wet = r["compound"] in WET or bool(r["rain"])
    return {
        "lap": int(r["lap"]),
        "laps_remaining": int(r["laps_remaining"]),
        "compound": r["compound"],
        "tyre_age_laps": None if age is None else int(age),
        "typical_stint_laps": t,
        "tyre_age_vs_typical": age_vs,
        "safety_car": bool(r["safety_car"]),
        "virtual_safety_car": bool(r["vsc"]),
        "rain": bool(r["rain"]),
        "gap_behind_s": None if gap is None else round(float(gap), 1),
        "pit_loss_s": round(float(pit_loss), 1),
        "stop_now_loses_position": gap is not None and gap < pit_loss,
        "stops_done": int(r["stops_done"]),
        "stop_still_required": int(r["stops_done"]) == 0 and not wet,
        "race_control_messages": list(r["messages"])[-5:],
    }
```

```python
"""Two baselines on the same facts: a hand-written rule, and a logistic regression."""
from sklearn.linear_model import LogisticRegression

AGE = {"well before": 0, "near": 1, "past": 2, "unknown": 0}


def rule_prob(f: dict) -> float:
    if f["laps_remaining"] < 3:
        return 0.0
    if (f["safety_car"] or f["virtual_safety_car"]) and f["tyre_age_vs_typical"] in ("near", "past"):
        return 1.0
    if f["tyre_age_vs_typical"] == "past" and (f["stop_still_required"] or f["laps_remaining"] > 8):
        return 1.0
    return 0.0


def features(f: dict) -> list:
    age = f["tyre_age_laps"] or 0
    typical = f["typical_stint_laps"] or 25
    return [
        age, age / typical, AGE[f["tyre_age_vs_typical"]], f["laps_remaining"],
        float(f["safety_car"]), float(f["virtual_safety_car"]), float(f["rain"]),
        f["gap_behind_s"] if f["gap_behind_s"] is not None else 60.0, f["pit_loss_s"],
        float(f["stop_now_loses_position"]), f["stops_done"], float(f["stop_still_required"]),
        float(f["compound"] == "SOFT"), float(f["compound"] == "MEDIUM"), float(f["compound"] == "HARD"),
    ]


def train_logreg(X, y):
    return LogisticRegression(max_iter=2000, class_weight="balanced").fit(X, y)


def logreg_prob(model, f: dict) -> float:
    return float(model.predict_proba([features(f)])[0][1])
```

```python
"""Scores against the teams' real stops. Pit laps are rare, so accuracy alone misleads."""
import numpy as np
import pandas as pd
from sklearn.metrics import average_precision_score, brier_score_loss, precision_recall_fscore_support


def lap_metrics(y, p, threshold: float) -> dict:
    pairs = [(int(a), float(b)) for a, b in zip(y, p) if b is not None and not (isinstance(b, float) and np.isnan(b))]
    missing = len(list(y)) - len(pairs)
    yy = np.array([a for a, _ in pairs])
    pp = np.array([b for _, b in pairs])
    pred = (pp >= threshold).astype(int)
    prec, rec, f1, _ = precision_recall_fscore_support(yy, pred, average="binary", zero_division=0)
    bins = np.clip((pp * 10).astype(int), 0, 9)
    ece = sum(abs(pp[bins == b].mean() - yy[bins == b].mean()) * (bins == b).sum() for b in range(10) if (bins == b).any()) / len(pp)
    return {
        "n": len(pp), "missing": missing, "positives": int(yy.sum()),
        "precision": round(float(prec), 3), "recall": round(float(rec), 3), "f1": round(float(f1), 3),
        "average_precision": round(float(average_precision_score(yy, pp)), 3) if yy.any() else None,
        "brier": round(float(brier_score_loss(yy, pp)), 4), "ece": round(float(ece), 4),
    }


def event_metrics(df: pd.DataFrame, prob_col: str, threshold: float, tolerance: int = 1) -> dict:
    found = total_stops = correct = total_calls = 0
    for _, g in df.groupby(["race", "driver"]):
        stops = set(g.loc[g["pitted"], "lap"])
        calls = set(g.loc[g[prob_col].fillna(-1) >= threshold, "lap"])
        near = lambda lap, s: any(abs(lap - x) <= tolerance for x in s)
        total_stops += len(stops)
        found += sum(1 for s in stops if near(s, calls))
        total_calls += len(calls)
        correct += sum(1 for c in calls if near(c, stops))
    return {
        "stops": total_stops, "calls": total_calls,
        "found_share": round(found / total_stops, 3) if total_stops else None,
        "correct_call_share": round(correct / total_calls, 3) if total_calls else None,
    }
```

```python
"""System 1: one Jev Noul for each driver and lap. Code calculates every fact first."""
import asyncio
from typesafe_sdk import AsyncTypeSafeClient, Noul, NoulCriteria
from .config import JEV_MODEL

INSTRUCTIONS = " ".join([
    "Must the driver in `lap` enter the pit lane at the end of this lap? Decide by these steps, in order.",
    "1. If `laps_remaining` is below 3, the answer is no.",
    "2. If `safety_car` or `virtual_safety_car` is true and `tyre_age_vs_typical` is near or past, the answer is yes: a stop now costs less time.",
    "3. If `tyre_age_vs_typical` is past and `stop_still_required` is true, the answer is yes.",
    "4. If `tyre_age_vs_typical` is past and `laps_remaining` is above 8, the answer is yes.",
    "5. If `stop_now_loses_position` is true and steps 2 to 4 do not apply, the answer is no.",
    "6. If a message in `race_control_messages` says that a safety car or a virtual safety car starts, treat it as step 2.",
    "7. Otherwise the answer is no.",
])


def pit_question() -> dict:
    return {"pit_now": Noul(
        instructions=INSTRUCTIONS,
        criteria=NoulCriteria(true="The driver must pit at the end of this lap.", false="The driver must stay out this lap."),
    )}


async def jev_probs(facts_list, concurrency: int = 8, client=None):
    own = client is None
    client = client or AsyncTypeSafeClient(model=JEV_MODEL)
    sem = asyncio.Semaphore(concurrency)
    usage = errors = 0
    questions = pit_question()

    async def one(f):
        nonlocal usage, errors
        async with sem:
            try:
                r = await client.system_one(state={"lap": f}, questions=questions)
                usage += getattr(getattr(r, "usage", None), "input_tokens", 0) or 0
                return float(r.nouls["pit_now"].noul)
            except Exception:  # The error is counted; the lap gets no probability.
                errors += 1
                return None

    try:
        probs = await asyncio.gather(*(one(f) for f in facts_list))
    finally:
        if own and hasattr(client, "close"):
            await client.close()
    jev_probs.last_usage, jev_probs.last_errors = usage, errors
    return list(probs)
```

```python
"""System 2: an LLM decides the uncertain laps. Code decides when, and code checks the answer."""
import json
import os
import re
import subprocess
import time

SC_WORDS = ("SAFETY CAR", "VIRTUAL SAFETY CAR", "VSC")


def should_escalate(p, f, low: float = 0.35, high: float = 0.65) -> bool:
    if p is None or low <= p <= high:
        return True
    return any(w in m.upper() for m in f["race_control_messages"] for w in SC_WORDS)


def parse_answer(text: str):
    first = re.sub(r"[^a-z]", "", (text or "").strip().split()[0].lower()) if (text or "").strip() else ""
    return 1.0 if first == "yes" else 0.0 if first == "no" else None


SYSTEM = "You are a Formula 1 race strategist. Answer with yes or no first, then one sentence."


def _prompt(f: dict) -> str:
    return "Facts about this driver and lap (calculated by code):\n" + json.dumps(f, indent=1) + \
        "\nMust the driver pit at the end of this lap? Answer yes or no first, then one sentence of reason."


def ask_llm(f: dict, backend: str = None) -> dict:
    backend = backend or os.environ.get("LLM_BACKEND", "anthropic")
    model = os.environ.get("LLM_MODEL", "claude-sonnet-5-5" if backend == "anthropic" else "sonnet")
    t0 = time.time()
    if backend == "claude-cli":
        r = subprocess.run(["claude", "-p", "--model", model, "--output-format", "json", "--system-prompt", SYSTEM],
                           input=_prompt(f), capture_output=True, text=True, timeout=120)
        d = json.loads(r.stdout) if r.returncode == 0 else {"result": "", "is_error": True}
        text = "" if d.get("is_error") else d.get("result", "")
    else:
        import anthropic
        msg = anthropic.Anthropic().messages.create(model=model, max_tokens=80, system=SYSTEM,
                                                    messages=[{"role": "user", "content": _prompt(f)}])
        text = "".join(c.text for c in msg.content if c.type == "text")
    return {"p": parse_answer(text), "text": text, "ms": round((time.time() - t0) * 1000)}
```

- [ ] **Step 1: Write `run.py`**

Create `f1/f1pit/run.py`:

```python
"""Command line for the F1 pit test.

  cd f1 && set -a && . ../.env && set +a
  .venv/bin/python -m f1pit.run train
  .venv/bin/python -m f1pit.run tune
  .venv/bin/python -m f1pit.run test --threshold 0.5 --system2 off
"""
import argparse
import asyncio
import json
import os
import pickle
import time
import pandas as pd
from .config import TRAIN_RACES, TUNE_RACES, TEST_RACES, RESULTS_DIR, DATA_DIR
from .data import load_race, pit_loss_s, typical_stints
from .facts import lap_facts
from .baselines import rule_prob, features, train_logreg, logreg_prob
from .jev import jev_probs
from .metrics import lap_metrics, event_metrics
from .system2 import should_escalate, ask_llm

JEV_PER_MTOK = 0.042
# Pickle is safe here: cmd_train() writes this file itself in f1/data/ (not committed). Never load a pickle from elsewhere.
MODEL_PATH = os.path.join(DATA_DIR, "model.pkl")


def facts_for(races, typical):
    rows = []
    for year, gp in races:
        t = load_race(year, gp)
        loss = pit_loss_s(t)
        for _, r in t.iterrows():
            rows.append({"race": r["race"], "driver": r["driver"], "lap": int(r["lap"]), "pitted": bool(r["pitted"]),
                         "facts": lap_facts(r, typical, loss)})
    return rows


def cmd_train():
    tables = [load_race(y, g) for y, g in TRAIN_RACES]
    typical = typical_stints(tables)
    rows = facts_for(TRAIN_RACES, typical)
    model = train_logreg([features(r["facts"]) for r in rows], [int(r["pitted"]) for r in rows])
    with open(MODEL_PATH, "wb") as fh:
        pickle.dump({"model": model, "typical": typical}, fh)
    print("typical stint lengths:", typical, "| training rows:", len(rows), "| stops:", sum(r["pitted"] for r in rows))


def score(rows, threshold, system2=False):
    df = pd.DataFrame([{k: v for k, v in r.items() if k != "facts"} for r in rows])
    out = {}
    for col in ("rule", "logreg", "jev", "final"):
        if col in df:
            out[col] = {**lap_metrics(df["pitted"], df[col].tolist(), threshold), **event_metrics(df, col, threshold)}
            sc = df[[r["facts"]["safety_car"] or r["facts"]["virtual_safety_car"] for r in rows]]
            if len(sc) and sc["pitted"].any():
                out[col]["safety_car_laps"] = lap_metrics(sc["pitted"], sc[col].tolist(), threshold)
    return out


def run_split(races, threshold, use_system2, label):
    with open(MODEL_PATH, "rb") as fh:
        saved = pickle.load(fh)
    rows = facts_for(races, saved["typical"])
    est = len(rows) * 650 * JEV_PER_MTOK / 1e6
    print(f"{len(rows)} driver-laps; Jev estimate ${est:.2f}")
    if est > 5:
        raise SystemExit("Estimate is over the $5 cap. Stop.")
    probs = asyncio.run(jev_probs([r["facts"] for r in rows]))
    s2_calls = s2_ms = 0
    for r, p in zip(rows, probs):
        r["rule"] = rule_prob(r["facts"])
        r["logreg"] = logreg_prob(saved["model"], r["facts"])
        r["jev"] = p
        r["final"] = p
        if use_system2 and should_escalate(p, r["facts"]):
            a = ask_llm(r["facts"])
            s2_calls += 1
            s2_ms += a["ms"]
            if a["p"] is not None:
                r["final"] = a["p"]
    result = {
        "label": label, "threshold": threshold, "system2": use_system2, "races": [f"{y} {g}" for y, g in races],
        "jev_input_tokens": jev_probs.last_usage, "jev_errors": jev_probs.last_errors,
        "jev_cost_usd": round(jev_probs.last_usage * JEV_PER_MTOK / 1e6, 4),
        "system2_calls": s2_calls, "system2_mean_ms": round(s2_ms / s2_calls) if s2_calls else None,
        "scores": score(rows, threshold),
    }
    os.makedirs(RESULTS_DIR, exist_ok=True)
    path = os.path.join(RESULTS_DIR, f"{label}-{time.strftime('%Y%m%dT%H%M%S')}.json")
    with open(path, "w") as fh:
        json.dump(result, fh, indent=1)
    print(json.dumps(result["scores"], indent=1))
    print("Results:", path)
    return result


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("step", choices=["train", "tune", "test"])
    ap.add_argument("--threshold", type=float, default=0.5)
    ap.add_argument("--system2", choices=["on", "off"], default="off")
    a = ap.parse_args()
    if a.step == "train":
        cmd_train()
    elif a.step == "tune":
        for t in (0.3, 0.4, 0.5, 0.6, 0.7):
            run_split(TUNE_RACES, t, a.system2 == "on", f"tune-t{t}")
    else:
        run_split(TEST_RACES, a.threshold, a.system2 == "on", f"test-s2{a.system2}")


if __name__ == "__main__":
    main()
```

Note: the `tune` step calls Jev once for each threshold. To save tokens, the executor may cache the probabilities from the first call and score the other thresholds from the cache; the result must be the same.
